\begin{s}\textbf{2} \bigskip\\
    Let $F$ be the event that an email has "free
    money", and let $S$ be the event that an email
    is spam.
    \begin{enumerate} 
        % 1
        \item 
            $P(F) = P(F | S) P(S) + P(F | \neg S) P(\neg S)$
            \bigskip\\
            $P(F) = \frac{1}{10} \cdot \frac{8}{10} +
            \frac{1}{100} \cdot \frac{2}{10}$ 
            \bigskip\\
            $P(F) = \frac{41}{500}$

        % 2
        \item
            $P(S | F) = \frac{P(F | S) P(S)}
            {P(F | S) P(S) + P(F | \neg S) P(\neg S)}$
            \bigskip\\
            $P(S | F) = \frac{\frac{1}{10} \cdot \frac{8}{10}}
            {\frac{1}{10} \cdot \frac{8}{10} + \frac{1}{100} \cdot \frac{2}{10}}$
            \bigskip\\
            $P(S | F) = \frac{40}{41}$

        % 3
        \item
            $P(\neg S |\neg  F) = \frac{P(\neg F | \neg S) P(\neg S)}
            {P(\neg F | \neg S) P(\neg S) + P(\neg F | S) P(S)}$
            \bigskip\\
            $P(\neg S | \neg F) = \frac{\frac{2}{10} \cdot \frac{99}{100}}
            {\frac{2}{10} \cdot \frac{99}{100} + \frac{8}{10} \cdot \frac{9}{10}}$
            \bigskip\\
            $P(\neg S | \neg F) = \frac{11}{51}$
    \end{enumerate}
\end{s}
