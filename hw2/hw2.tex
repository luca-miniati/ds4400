\begin{s}\textbf{6}
    \begin{enumerate}
        \item
            \begin{align*}
                J(\theta) &= (X\theta - y)^T(X\theta - y) + \lambda \theta^T\theta \\
                          &= (\theta^TX^T - y^T)(X\theta - y) + \lambda \theta^T\theta \\
                          &= \theta^TX^TX\theta - y^TX\theta - y \theta^TX^T + y^Ty+ \lambda \theta^T\theta \\
                          &= \theta^TX^TX\theta - 2X^Ty\theta + y^Ty+ \lambda \theta^T\theta \\
                \nabla J &= 2X^TX\theta - 2X^Ty + 2\lambda\theta \\
                        0 &= 2X^TX\theta - 2X^Ty + 2\lambda\theta \\
                        2X^Ty &= 2(X^TX + I\lambda)\theta \\
                        \theta &= (X^TX + I\lambda)^{-1}X^Ty
            \end{align*}
